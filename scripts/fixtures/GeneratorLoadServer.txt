}
function generatorLoadMeter_(run,fn) {
  generatorLoadGuard_();
  var p=PropertiesService.getScriptProperties(),key='GEN_TEST_'+run;
  finalLock_(function(){var s=JSON.parse(p.getProperty(key));s.active++;s.peak=Math.max(s.peak,s.active);p.setProperty(key,JSON.stringify(s));});
  try{return fn();}finally{finalLock_(function(){var s=JSON.parse(p.getProperty(key));s.active--;p.setProperty(key,JSON.stringify(s));});}
}

function generatorLoadVerify(run,jobs) {
  generatorLoadGuard_();
  var p=PropertiesService.getScriptProperties(),seen={}, docs={}, rows=reservationRows_(getRequiredSheet_(APP.DATA_SHEET),REQUEST_HEADERS.length);
  var results=jobs.map(function(input){
    var job=JSON.parse(p.getProperty('LETTER_DRAFT_'+input.token)||'{}');
    var matching=rows.filter(function(r){return r[15]===input.token;});
    var issues=[];
    if(matching.length!==1) issues.push('register count '+matching.length);
    var r=matching[0]||[], unique=r[2]+'/'+r[12]+'/'+r[1];
    if(seen[unique]) issues.push('duplicate number');seen[unique]=true;
    if(docs[job.docId]) issues.push('duplicate document');docs[job.docId]=true;
    if(r[10]!==REQUEST_STATUS.ACTIVE||!r[14]) issues.push('PDF archive/status missing');
    if(!job.numberedResult) issues.push('numbered result missing');
    if(job.docId && job.numberedResult) {
      var text=DocumentApp.openById(job.docId).getBody().getText();
      if(text.indexOf(job.numberedResult.code)<0||text.indexOf('[BELUM BERNOMOR]')>=0) issues.push('document number mismatch');
      if(text.indexOf(input.tag)<0 && input.data.type!=='GEDUNG') issues.push('wrong document content');
      if(DriveApp.getFileById(r[14]).getMimeType()!==MimeType.PDF) issues.push('archive is not PDF');
    }
    return {type:input.data.type,number:r[1],ok:!issues.length,issues:issues};
  });
  var stats=JSON.parse(p.getProperty('GEN_TEST_'+run));
  return {run:run,peak:stats.peak,active:stats.active,results:results};
}
